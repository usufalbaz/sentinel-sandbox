function severityColor(severity: Severity): string {
  switch (severity) {
    case "CRITICAL":
    case "HIGH":
      return "#E24B4A";
    case "MEDIUM":
      return "#EF9F27";
    case "LOW":
    case "INFO":
    default:
      return "#697565";
  }
}

}

interface Message {
  role: "user" | "assistant";
  text: string;
}

const VERDICT_TINT: Record<VerdictLevel, { border: string; bg: string }> = {
  SAFE: { border: "#97C459", bg: "#0f1e04" },
  SUSPICIOUS: { border: "#EF9F27", bg: "#1a0f02" },
  DANGEROUS: { border: "#E24B4A", bg: "#1f0808" },
};

export default function FollowUpChat({
  scanId,
  verdictLevel,
}: {
  scanId: string;
  verdictLevel?: VerdictLevel;
}) {
  const [messages, setMessages] = useState<Message[]>([]);
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);
  const bottomRef = useRef<HTMLDivElement>(null);

  const tint = VERDICT_TINT[verdictLevel ?? "SAFE"];

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages]);

  async function handleAsk(e: React.FormEvent) {
    e.preventDefault();
    const q = input.trim();
    if (!q || loading) return;
    setInput("");
    setMessages((prev) => [...prev, { role: "user", text: q }]);
    setLoading(true);
    try {
      const { answer } = await askFollowUp(scanId, q);
      setMessages((prev) => [...prev, { role: "assistant", text: answer }]);
    } catch {
      setMessages((prev) => [
        ...prev,
        { role: "assistant", text: "Error: could not get a response." },
      ]);
    } finally {
      setLoading(false);
    }
  }

  return (
    <div
      className="w-full flex flex-col"
      style={{
        background: "#1A1D16",
        border: "1px solid #3E3F38",
        borderRadius: "2px",

          <p className="text-xs font-mono" style={{ color: "#697565" }}>
            ask anything about this scan…
          </p>
        )}
        {messages.map((msg, i) => (
          <div
            key={i}
            className={`max-w-[85%] px-3 py-2 text-sm ${

            }`}
            style={
              msg.role === "user"
                ? {
                    background: "#2E2F29",
                    color: "#ECDFCC",
                    borderRadius: "2px",
                  }
                : {
                    background: tint.bg,
                    border: `1px solid ${tint.border}`,
                    color: "#A8A596",
                    borderRadius: "2px",
                  }
            }
          >
            {msg.text}
          </div>
        ))}
        {loading && (
          <div
            className="self-start px-3 py-2 text-sm font-mono"
            style={{
              background: tint.bg,
              border: `1px solid ${tint.border}`,
              color: "#697565",
              borderRadius: "2px",
            }}
          >
            thinking…
          </div>

            background: "#ECDFCC",
            color: "#0F120D",
            borderRadius: "2px",
            border: "none",
            cursor: loading || !input.trim() ? "not-allowed" : "pointer",
          }}
        >
          Ask
        </button>
      </form>
    </div>
  );
}

export default function Home() {
  const [scanId, setScanId] = useState<string | null>(null);
  const [scan, setScan] = useState<ScanRecord | null>(null);
  const [narrative, setNarrative] = useState<string | undefined>(undefined);
  const [narrativeLoading, setNarrativeLoading] = useState(false);

  // ── Polling ──────────────────────────────────────────────────────────────
  const poll = useCallback(async (id: string) => {
    try {
      const data = await getScan(id);
      setScan(data);
      return data.status;
    } catch {
      return "ERROR";
    }
  }, []);

  useEffect(() => {
    if (!scanId) return;
    // immediate first fetch
    poll(scanId);
    const iv = setInterval(async () => {
      const status = await poll(scanId);
      if (status === "COMPLETE" || status === "ERROR") {
        clearInterval(iv);
      }
    }, 2000);
    return () => clearInterval(iv);
  }, [scanId, poll]);

  // ── Fetch narrative once scan is COMPLETE ─────────────────────────────
  useEffect(() => {
    if (!scan || scan.status !== "COMPLETE" || narrative || narrativeLoading)
      return;
    setNarrativeLoading(true);
    getNarrative(scan.scan_id)
      .then(({ narrative: n }) => setNarrative(n))
      .catch(() => {/* narrative is optional, ignore errors */})
      .finally(() => setNarrativeLoading(false));

function repoShort(url: string): string {
  try {
    const parts = new URL(url).pathname.replace(/^\//, "").split("/");
    return parts.slice(0, 2).join("/");
  } catch {
    return url.length > 32 ? url.slice(-32) : url;
  }
}

        const isActive = scan.scan_id === activeScanId;

const STATUS_LABEL: Record<string, string> = {
  QUEUED:           "Queued",
  CLONING:          "Cloning repository…",
  SCANNING_STATIC:  "Running static analysis…",
  RUNNING_SANDBOX:  "Sandbox executing…",
  COMPLETE:         "Complete",
  ERROR:            "Error",
};

const VERDICT_COLOR: Record<VerdictLevel, string> = {
  SAFE: "#4caf8d",
  SUSPICIOUS: "#d4943a",
  DANGEROUS: "#992b2b",
};

type Tab = "overview" | "findings" | "chat";

// ── Main page ─────────────────────────────────────────────────────────────
export default function Home() {
  // All scans this session (in-order, append-only)
  const [scans, setScans] = useState<ScanRecord[]>([]);
  // The ID being actively viewed/polled
  const [activeScanId, setActiveScanId] = useState<string | null>(null);
  // Narrative for the active scan
  const [narrative, setNarrative] = useState<string | undefined>(undefined);
  const [narrativeLoading, setNarrativeLoading] = useState(false);
  // Dashboard tab
  const [tab, setTab] = useState<Tab>("overview");
  // Sidebar view: "new" = new scan form, "history" = scan history
  const [sidePanel, setSidePanel] = useState<"new" | "history">("new");

  // Derived: the active scan record
  const activeScan = scans.find((s) => s.scan_id === activeScanId) ?? null;

  // ── Upsert a scan record into the list ──────────────────────────────────
  const upsertScan = useCallback((updated: ScanRecord) => {
    setScans((prev) => {
      const idx = prev.findIndex((s) => s.scan_id === updated.scan_id);
      if (idx === -1) return [...prev, updated];
      const next = [...prev];
      next[idx] = updated;
      return next;
    });
  }, []);

  // ── Poll the active scan ─────────────────────────────────────────────────
  const poll = useCallback(
    async (id: string) => {
      try {
        const data = await getScan(id);
        upsertScan(data);
        return data.status;
      } catch {
        return "ERROR";
      }
    },
    [upsertScan]
  );

  useEffect(() => {
    if (!activeScanId) return;
    poll(activeScanId);
    const iv = setInterval(async () => {
      const status = await poll(activeScanId);
      if (status === "COMPLETE" || status === "ERROR") clearInterval(iv);
    }, 2000);
    return () => clearInterval(iv);
  }, [activeScanId, poll]);

  // ── Fetch narrative once COMPLETE ───────────────────────────────────────
  useEffect(() => {
    if (
      !activeScan ||
      activeScan.status !== "COMPLETE" ||
      narrative ||
      narrativeLoading
    )
      return;
    setNarrativeLoading(true);
    getNarrative(activeScan.scan_id)
      .then(({ narrative: n }) => setNarrative(n))
      .catch(() => {})
      .finally(() => setNarrativeLoading(false));
  }, [activeScan, narrative, narrativeLoading]);

  // ── Handle new scan started ──────────────────────────────────────────────
  function handleScanStarted(scanId: string) {
    setActiveScanId(scanId);
    setNarrative(undefined);
    setNarrativeLoading(false);
    setTab("overview");
    setSidePanel("history");
  }

  // ── Select a scan from history ──────────────────────────────────────────
  function handleSelectScan(scanId: string) {
    if (scanId === activeScanId) return;
    setActiveScanId(scanId);
    setNarrative(undefined);
    setNarrativeLoading(false);
    setTab("overview");
  }

  // ── Reset to idle ────────────────────────────────────────────────────────
  function startNewScan() {
    setActiveScanId(null);
    setNarrative(undefined);
    setNarrativeLoading(false);
    setTab("overview");
    setSidePanel("new");
  }

  // ── Derived booleans ─────────────────────────────────────────────────────
  const isIdle     = !activeScanId;
  const isPolling  = activeScan && activeScan.status !== "COMPLETE" && activeScan.status !== "ERROR";
  const isComplete = activeScan?.status === "COMPLETE";
  const isError    = activeScan?.status === "ERROR";

  // ── Render ────────────────────────────────────────────────────────────────
  return (
    <div
      className="flex flex-col min-h-screen"
      style={{ background: C.bg, color: C.text }}
    >
      {/* ── Top nav bar ── */}
      <header
        className="flex items-center gap-3 px-5 py-3 shrink-0"
        style={{ background: C.surface, borderBottom: `1px solid ${C.border}` }}
      >
        <LogoMark />
        <span className="text-sm font-mono font-semibold tracking-tight" style={{ color: C.text }}>
          sentinel<span style={{ color: C.accent }}>-sandbox</span>
        </span>

        <div className="flex-1" />

        {/* Status pill (while scanning) */}
        {isPolling && activeScan && (
          <div
            className="flex items-center gap-2 px-3 py-1 text-xs font-mono"
            style={{ background: C.bg, border: `1px solid ${C.border}` }}
          >
            <span className="pulse-dot" />
            {STATUS_LABEL[activeScan.status] ?? activeScan.status}
          </div>
        )}

        {/* Active verdict pill */}
        {isComplete && activeScan?.verdict && (
          <div
            className="flex items-center gap-2 px-3 py-1 text-xs font-mono font-bold"
            style={{
              background:
                activeScan.verdict.level === "DANGEROUS"
                  ? C.dangerBg
                  : activeScan.verdict.level === "SUSPICIOUS"
                  ? "#2e1f06"
                  : "#0b2a1e",
              color: VERDICT_COLOR[activeScan.verdict.level],
              border: `1px solid ${VERDICT_COLOR[activeScan.verdict.level]}`,
            }}
          >
            {activeScan.verdict.level}
          </div>
        )}

        {/* New scan button */}
        {!isIdle && (
          <button
            onClick={startNewScan}
            className="px-3 py-1 text-xs font-mono"
            style={{
              background: C.raised,
              color: C.text,
              border: `1px solid #1f4e90`,

            }}
          >
            + new scan
          </button>
        )}
      </header>

      {/* ── Body: sidebar + main ── */}
      <div className="flex flex-1 min-h-0">

        {/* ── Sidebar ── */}
        <aside
          className="flex flex-col shrink-0"
          style={{
            width: "220px",
            background: C.surface,
            borderRight: `1px solid ${C.border}`,
          }}
        >
          {/* Nav items */}
          <nav className="flex flex-col pt-2 pb-1">
            <div
              className={`nav-item${tab === "overview" && !isIdle ? " active" : ""}`}
              onClick={() => !isIdle && setTab("overview")}
              style={{ opacity: isIdle ? 0.4 : 1 }}
            >
              <IconDashboard /> overview
            </div>
            <div
              className={`nav-item${tab === "findings" && !isIdle ? " active" : ""}`}
              onClick={() => !isIdle && setTab("findings")}
              style={{ opacity: isIdle ? 0.4 : 1 }}
            >
              <IconFindings />
              findings
              {isComplete && activeScan && (
                <span

                  style={{ color: C.accent }}
                >
                  {activeScan.static_findings.length +
                    activeScan.runtime_findings.length}
                </span>
              )}
            </div>
            <div
              className={`nav-item${tab === "chat" && !isIdle ? " active" : ""}`}
              onClick={() => !isIdle && setTab("chat")}
              style={{ opacity: isIdle ? 0.4 : 1 }}
            >
              <IconChat /> follow-up
            </div>
          </nav>

          {/* Divider */}
          <div style={{ height: 1, background: C.border, margin: "4px 0" }} />

          {/* Scan section */}
          <nav className="flex flex-col pb-1">
            <div

              onClick={() => setSidePanel("new")}

            >

          {/* Side panel content */}
          <div className="flex-1 min-h-0 overflow-hidden flex flex-col">
            {sidePanel === "new" ? (
              <div className="p-3 flex flex-col gap-2">
                <span

                  scan a repo
                </span>
                <ScanForm onScanStarted={handleScanStarted} />
              </div>
            ) : (
              <ScanHistory
                scans={scans}

                onSelect={handleSelectScan}

          </div>
        </aside>

        {/* ── Main content ── */}

                <svg

                >
                  <filter id="grain">
                    <feTurbulence type="fractalNoise" baseFrequency="0.9" numOctaves={3} stitchTiles="stitch" />
